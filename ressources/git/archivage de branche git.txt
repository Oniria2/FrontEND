# Create a tag named similarly to the branch but with the 'LTS/' prefix
# Point it to the latest commit of the branch

git tag LTS/<branchname> <branchname>

# Push local tags to remote

git push --tags

#go back to main

git checkout main

# efface la branche si elle existe et la recrée puis bascule dessus

git checkout -b nom_de_la_branche

# bascule sur la branche

git checkout nom_de_la_branche

# Delete the branch locally

git branch -D <branchname>

# Delete the local reference of remote branch ("forget" the remote branch)

git branch -d -r origin/<branchname>

# Delete the remote branch

git push -d origin <branchname>

# or, similarly, 'git push origin :<branchname>'
